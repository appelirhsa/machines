OUTCOME A below.

DC Machines (Outcome A)

\[ E=\frac{\phi ZN}{60}\left(\frac{P}{A}\right) \qquad \text{Lap: }A=P,\ \text{Wave: }A=2 \]
\[ P_d=EI_a=T_e\omega_m \qquad T_e=k_a\phi I_a \qquad \omega_m=\frac{2\pi N}{60} \]
\[ V=E+I_aR_a\ (\text{motor}) \qquad E=V-I_aR_a\ (\text{back-emf}) \qquad E=V+I_aR_a\ (\text{generator}) \]
\[ n\propto\frac{E}{\phi} \qquad (\phi\propto I_a\Rightarrow T\propto I_a^2) \qquad (\phi=\text{const}\Rightarrow E\propto n) \]
\[ \eta=\frac{P_{\text{out}}}{P_{\text{in}}} \qquad I_a^2R_a=P_{\text{constant}} \Rightarrow I_a=\sqrt{\frac{P_c}{R_a}} \]

Induction Motors (Outcome B)

\[ n_s=\frac{120f}{p} \qquad s=\frac{n_s-n}{n_s} \qquad n=(1-s)n_s \qquad f_2=sf_1 \]
\[ P_g=3I_2^2\left(\frac{R_2}{s}\right) \qquad P_{\text{Rotor Cu}}=sP_g \qquad P_{\text{dev}}=(1-s)P_g \]
\[ T_e=\frac{P_g}{\omega_s} \qquad R_2=sX_2 \qquad \frac{T}{T_{\max}}=\frac{2(s^\*/s)}{1+(s^\*/s)^2} \]
\[ R_e=\frac{P_s}{I_s^2}, \qquad X_e=\frac{\sqrt{(V_sI_s)^2-P_s^2}}{I_s^2} \]
\[ \frac{T_s}{T_{FL}}=\left(\frac{kI_{s,\text{full}}}{I_{FL}}\right)^2s_{FL} \qquad \text{Star-delta: }I,T\approx\frac{1}{3}\text{ of DOL} \]

Synchronous Machines (Outcome B)

\[ n_s=\frac{120f}{p} \qquad q=\frac{Q}{pm} \qquad \alpha=\frac{180^\circ}{mq} \qquad k_d=\frac{\sin(q\alpha/2)}{q\sin(\alpha/2)} \]
\[ k_p=\sin\left(\frac{90^\circ\beta}{\tau}\right) \qquad k_w=k_dk_p \qquad E_{\mathrm{rms}}=4.44k_wfN\phi_m \]
\[ V_0=V_t+I_a(R_a+jX_s) \qquad \%\text{reg}=\frac{V_0-V_t}{V_t}\times100 \]
\[ P=\left(\frac{V_tV_0}{X_s}\right)\sin\delta \]
\[ P=\left(\frac{V_tV_0}{X_d}\right)\sin\delta+\left(\frac{V_t^2}{2}\right)\left(\frac{1}{X_q}-\frac{1}{X_d}\right)\sin2\delta \]
\[ \text{Over-excited}\Rightarrow\text{leading PF (supplies vars)} \qquad \text{Under-excited}\Rightarrow\text{lagging PF (absorbs vars)} \]

Mechanics & Energy Conversion (Outcome A & B applications)

\[ P=T\omega \qquad \omega=\frac{2\pi N}{60} \qquad T=J\left(\frac{d\omega}{dt}\right)+b\omega+T_{\text{load}} \]
\[ N_1D_1=N_2D_2 \qquad \frac{T_{\text{out}}}{T_{\text{in}}}=\frac{N_{\text{in}}}{N_{\text{out}}} \]
\[ T_{\text{motor}}=\frac{(T_{\text{load}}/\text{ratio})}{\eta_{gb}} \qquad P_{\text{out}}=P_{\text{in}}\eta_{gb} \]
\[ F=mg \qquad T_{\text{drum}}=Fr \qquad P=Fv \qquad v=\frac{\text{distance}}{\text{time}} \]
\[ P_{\text{source}}=\frac{P_{\text{load}}}{\eta_1\eta_2\eta_3\cdots} \qquad \text{Runtime (h)}\approx\frac{\text{Capacity (A·h)}}{\text{Current (A)}} \]
\[ P_{\text{hyd}}=\frac{\rho Vgh}{t},\ \rho_{\text{water}}\approx1000\,\mathrm{kg/m^3} \qquad P=\sqrt{3}V_LI_L\cos\phi \]
